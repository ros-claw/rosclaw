cmake_minimum_required(VERSION 3.16)
project(rosclaw_tf_attribution LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(rosclaw_tf_attribution tf_attribution.cpp)
target_compile_features(rosclaw_tf_attribution PRIVATE cxx_std_17)
target_compile_options(rosclaw_tf_attribution PRIVATE -Wall -Wextra -Wpedantic -Werror)
ament_target_dependencies(rosclaw_tf_attribution rclcpp tf2_msgs std_msgs)
install(TARGETS rosclaw_tf_attribution DESTINATION lib/${PROJECT_NAME})
