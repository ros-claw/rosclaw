cmake_minimum_required(VERSION 3.16)
project(rosclaw_passive_physics LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(rosclaw_passive_physics SHARED passive_physics.cc)
target_compile_features(rosclaw_passive_physics PRIVATE cxx_std_17)
target_compile_options(rosclaw_passive_physics PRIVATE -Wall -Wextra -Werror)
target_link_libraries(rosclaw_passive_physics PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
add_library(rosclaw_passive_contacts SHARED passive_contacts.cc)
target_compile_features(rosclaw_passive_contacts PRIVATE cxx_std_17)
target_compile_options(rosclaw_passive_contacts PRIVATE -Wall -Wextra -Werror)
target_link_libraries(rosclaw_passive_contacts PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
include(CTest)
if(BUILD_TESTING)
  add_executable(passive_physics_contract_test contract_test.cc)
  target_compile_features(passive_physics_contract_test PRIVATE cxx_std_17)
  target_compile_options(passive_physics_contract_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(passive_physics_contract_test PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
  add_test(NAME passive_ecm_contract COMMAND passive_physics_contract_test)
  add_executable(passive_contacts_contract_test contact_contract_test.cc)
  target_compile_features(passive_contacts_contract_test PRIVATE cxx_std_17)
  target_compile_options(passive_contacts_contract_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(passive_contacts_contract_test PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
  add_test(NAME passive_contacts_ecm_contract COMMAND passive_contacts_contract_test)
endif()
